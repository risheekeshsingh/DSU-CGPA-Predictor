  const CreatorCard = ({ name, role, imgData, linkedinUrl }) => {

      const rect = cardRef.current.getBoundingClientRect();
      const x = e.clientX - rect.left;
      const y = e.clientY - rect.top;

      const centerX = rect.width / 2;
      const centerY = rect.height / 2;
      
      const rotateX = ((y - centerY) / centerY) * -10;
      const rotateY = ((x - centerX) / centerX) * 10;

      setTransform(
        `perspective(800px) rotateX(${rotateX}deg) rotateY(${rotateY}deg) scale3d(1.02,1.02,1.02)`
      );
      setGlare({ x: (x / rect.width) * 100, y: (y / rect.height) * 100, opacity: 0.1 });
    };

    const handleMouseLeave = () => {
      setTransform("perspective(800px) rotateX(0deg) rotateY(0deg) scale3d(1,1,1)");
      setGlare({ x: 50, y: 50, opacity: 0 });
    };

    return (
      <div
        ref={cardRef}
        onMouseMove={handleMouseMove}
        onMouseLeave={handleMouseLeave}
        style={{
          transform,
          transition: "transform 0.15s ease-out",
          transformStyle: "preserve-3d",
          willChange: "transform",

      >

          <img src={imgData} alt={name} className="w-full h-full object-cover object-top" />
        </div>
        
        <div className="text-center relative z-10">
          <h3 className="text-xl font-extrabold text-slate-900 m-0 tracking-tight">{name}</h3>
          <p className="text-sm text-slate-500 mt-1 font-medium">{role}</p>
        </div>
        
        <a href={linkedinUrl} target="_blank" rel="noopener noreferrer"
          onClick={(e) => e.stopPropagation()}
          className="mt-2 inline-flex items-center gap-2 bg-slate-50 hover:bg-blue-50 text-blue-700 px-6 py-2.5 rounded-full text-sm font-bold no-underline border border-slate-200 hover:border-blue-200 transition-colors z-10 shadow-sm">

    { name:"Risheekesh Singh", role:"Creator", imgData:`data:image/png;base64,${IMG2}`, linkedinUrl:"https://l1nk.dev/y684myw" },
    { name:"Aayush Bajaj", role:"Creator", imgData:`data:image/png;base64,${IMG3}`, linkedinUrl:"https://www.linkedin.com/in/aayush-bajaj-b3b97a344?utm_source=share_via&utm_content=profile&utm_medium=member_android" },
  ];

          <div className="flex flex-col md:flex-row items-start md:items-center justify-between mb-10 gap-6">

                            </div>
                            <div>

                      ) : (

                          onChange={(e) => updateSemester(sem.id, "sgpa", e.target.value)}
                          className="w-full p-3 rounded-xl bg-slate-50 border border-slate-200 focus:border-blue-500 focus:ring-2 focus:ring-blue-500/20 transition-all text-slate-900 outline-none" placeholder="e.g. 8.5" />
                      </div>
                      <div>
                        <label className="block mb-2 text-sm font-medium text-slate-600">Total Credits</label>
                        <input type="number" min="1" value={sem.credits}
                          onChange={(e) => updateSemester(sem.id, "credits", e.target.value)}
